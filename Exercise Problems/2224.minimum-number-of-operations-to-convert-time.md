public class Solution {
    public int ConvertTime(string current, string correct) {
        var currentMinutes = 
            (current[0] - '0') * 600 + 
            (current[1] - '0') * 60 + 
            (current[3] - '0') * 10 + 
            (current[4] - '0');
        var correctMinutes = 
            (correct[0] - '0') * 600 + 
            (correct[1] - '0') * 60 + 
            (correct[3] - '0') * 10 + 
            (correct[4] - '0');
        
        var diff = correctMinutes - currentMinutes;
        var ops = 0;

        ops += diff / 60;
        diff %= 60;
        
        ops += diff / 15;
        diff %= 15;

        ops += diff / 5;
        diff %= 5;

        ops += diff;

        return ops;
    }
}

class Solution:
    def convertTime(self, current: str, correct: str) -> int:
        current_minutes = (ord(current[0]) - ord('0')) * 600 + (ord(current[1]) - ord('0')) * 60 + (ord(current[3]) - ord('0')) * 10 + (ord(current[4]) - ord('0'))
        correct_minutes = (ord(correct[0]) - ord('0')) * 600 + (ord(correct[1]) - ord('0')) * 60 + (ord(correct[3]) - ord('0')) * 10 + (ord(correct[4]) - ord('0'))

        diff = correct_minutes - current_minutes
        ops = 0

        ops += diff // 60
        diff %= 60

        ops += diff // 15
        diff %= 15

        ops += diff // 5
        diff %= 5

        ops += diff

        return ops

```
class Solution:
    def convertTime(self, current: str, correct: str) -> int:
        time1 = int(current[:2]) * 60 + int(current[3:])
        time2 = int(correct[:2]) * 60 + int(correct[3:])
        diff = time2 - time1
        res = 0
        
        for t in [60, 15, 5, 1]:
            res += diff // t
            diff %= t
        return res
```
