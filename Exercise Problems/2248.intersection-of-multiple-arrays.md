**https://leetcode.cn/problems/intersection-of-multiple-arrays/description/**

- Solution by enumeration
```
public class Solution {
    public IList<int> Intersection(int[][] nums) {
        var ans = new List<int>();
        var firstArr = nums[0];
        foreach (var item in firstArr)
        {
            var flag = true;
            for (var i = 1; i < nums.Length; i++)
            {
                if (!nums[i].Contains(item))
                {
                    flag = false;
                    break;
                }
            }
            if (flag)
                ans.Add(item);
        }
        return ans.OrderBy(i => i).ToList();
    }
}
```
**Python**
```
class Solution:
    def intersection(self, nums: List[List[int]]) -> List[int]:
        ans = []
        first_arr = nums[0]
        for item in first_arr:
            flag = True
            for i in range(1, len(nums)):
                if item not in nums[i]:
                    flag = False
                    break
            if flag:
                ans.append(item)
        ans.sort()
        return ans
```

- Recommendation 1 by simulation
```
class Solution:
    def intersection(self, nums: List[List[int]]) -> List[int]:
        n = len(nums)
        res = set(nums[0])
        for i in range(1, n):
            tmp = set()
            for num in nums[i]:
                if num in res:
                    tmp.add(num)
            res = tmp
        return sorted(res)
```

- Recommendation 2 by counting freq
```
class Solution:
    def intersection(self, nums: List[List[int]]) -> List[int]:
        n = len(nums)
        freq = defaultdict(int)
        for arr in nums:
            for num in arr:
                freq[num] += 1
        res = []
        for k, v in freq.items():
            if v == n:
                res.append(k)
        return sorted(res)
```
