public class Solution {
    public IList<string> RemoveAnagrams(string[] words) {
        var result = new List<string>(words);
        var idx = 1;
        while (idx < result.Count)
        {
            var leftArr = result[idx - 1].ToCharArray();
            Array.Sort(leftArr);
            var rightArr = result[idx].ToCharArray();
            Array.Sort(rightArr);
            if (new string(leftArr) == new string(rightArr))
                result.RemoveAt(idx);
            else
                idx++;
        }
        return result;
    }
}

**Python**
```
class Solution:
    def removeAnagrams(self, words: List[str]) -> List[str]:
        result = list(words)
        idx = 1
        while idx < len(result):
            left_arr = [c for c in result[idx - 1]]
            left_arr.sort()
            right_arr = [c for c in result[idx]]
            right_arr.sort()
            if "".join(left_arr) == "".join(right_arr):
                result.pop(idx)
            else:
                idx += 1
        return result
```

- Recommendation by checking gradually
```
class Solution:
    def removeAnagrams(self, words: List[str]) -> List[str]:
        res = [words[0]]
        n = len(words)
        
        def compare(word1: str, word2: str) -> bool:
            freq = [0] * 26
            for ch in word1:
                freq[ord(ch) - ord('a')] += 1
            for ch in word2:
                freq[ord(ch) - ord('a')] -= 1
            return all(x == 0 for x in freq)

        for i in range(1, n):
            if compare(words[i], words[i - 1]):
                continue
            res.append(words[i])
        return res
```
