**https://leetcode.cn/problems/find-the-difference-of-two-arrays/description/**

- Solution by enumeration
```
public class Solution {
    public IList<IList<int>> FindDifference(int[] nums1, int[] nums2) {
        var res = new List<IList<int>>();
        var set1 = new HashSet<int>();
        var set2 = new HashSet<int>();
        foreach (var n1 in nums1)
        {
            if (!nums2.Contains(n1))
                set1.Add(n1);
        }
        foreach (var n2 in nums2)
        {
            if (!nums1.Contains(n2))
                set2.Add(n2);
        }
        res.Add(set1.ToList());
        res.Add(set2.ToList());
        return res;
    }
}
```
**Python**
```
class Solution:
    def findDifference(self, nums1: List[int], nums2: List[int]) -> List[List[int]]:
        res = []
        set1 = set()
        set2 = set()
        for n1 in nums1:
            if n1 not in nums2:
                set1.add(n1)
        for n2 in nums2:
            if n2 not in nums1:
                set2.add(n2)
        res.append(list(set1))
        res.append(list(set2))
        return res

```

- Recommendation by hashset collection
```
class Solution:
    def findDifference(self, nums1: List[int], nums2: List[int]) -> List[List[int]]:
        set1 = set(nums1)
        set2 = set(nums2)
        res = [[], []]
        for num in set1:
            if num not in set2:
                res[0].append(num)
        for num in set2:
            if num not in set1:
                res[1].append(num)
        return res

```
