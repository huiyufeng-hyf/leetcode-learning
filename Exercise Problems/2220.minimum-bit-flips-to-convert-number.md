**https://leetcode.cn/problems/minimum-bit-flips-to-convert-number/description/**

- Solution by XOR calculation
```
public class Solution {
    public int MinBitFlips(int start, int goal) {
        var diff = (uint)(start ^ goal);
        var flips = 0;
        while (diff != 0)
        {
            flips += (int)(diff & 1);
            diff >>= 1;
        }
        return flips;
    }
}
```

class Solution:

        tmp = start ^ goal
        while tmp:
            res += tmp & 1
            tmp >>= 1
        return res
