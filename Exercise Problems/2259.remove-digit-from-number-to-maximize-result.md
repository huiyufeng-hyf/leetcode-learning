- Solution by simulation
```
public class Solution {
    public string RemoveDigit(string number, char digit) {
        var n = number.Length;
        var lastIdx = -1;
        for (var i = 0; i < n; i++)
        {
            if (digit == number[i])
            {
                lastIdx = i;
                if (i + 1 < n && number[i + 1] > digit)
                    return number[..i] + number[(i + 1)..];
            }
        }
        return number[..lastIdx] + number[(lastIdx + 1)..];
    }
}
```
**Python**
```
class Solution:
    def removeDigit(self, number: str, digit: str) -> str:
        n, last_idx = len(number), -1
        for i in range(n):
            if number[i] == digit:
                last_idx = i
                if i + 1 < n and number[i + 1] > digit:
                    return number[:i] + number[i + 1:]
        return number[:last_idx] + number[(last_idx + 1):]
```

```
class Solution:
    def removeDigit(self, number: str, digit: str) -> str:
        n = len(number)
        res = ""
        for i in range(n):
            if number[i] == digit:
                tmp = number[:i] + number[i+1:]
                res = max(res, tmp)
        return res
```

```
class Solution:
    def removeDigit(self, number: str, digit: str) -> str:
        n = len(number)
        last = -1
        for i in range(n):
            if number[i] == digit:
                last = i
                if i < n - 1 and number[i] < number[i+1]:
                    return number[:i] + number[i+1:]
        return number[:last] + number[last+1:]
```
