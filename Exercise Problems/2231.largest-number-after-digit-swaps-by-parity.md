**https://leetcode.cn/problems/largest-number-after-digit-swaps-by-parity/description/**

- Solution by enumeration
```
public class Solution {
    public int LargestInteger(int num) {
        var digits = num.ToString().ToCharArray();
        var n = digits.Length;
        for (var i = 0; i < n; i++)
        {
            var leftDigit = digits[i] - '0';
            var bestIndex = i;
            for (var j = i + 1; j < n; j++)
            {
                var rightDigit = digits[j] - '0';
                var bestDigit = digits[bestIndex] - '0';
                if (rightDigit % 2 == leftDigit % 2 && rightDigit > bestDigit)
                    bestIndex = j;
            }
            if (bestIndex != i)
            {
                var temp = digits[i];
                digits[i] = digits[bestIndex];
                digits[bestIndex] = temp;
            }
        }
        return Convert.ToInt32(new string(digits));
    }
}
```
**Python**
```
class Solution:
    def largestInteger(self, num: int) -> int:
        digits = list(str(num))
        n = len(digits)
        for i in range(n):
            left_digit = ord(digits[i]) - ord('0')
            best_index = i
            for j in range(i + 1, n):
                right_digit = ord(digits[j]) - ord('0')
                best_digit = ord(digits[best_index]) - ord('0')
                if right_digit % 2 == left_digit % 2 and right_digit > best_digit:
                    best_index = j
            if best_index != i:
                digits[i], digits[best_index] = digits[best_index], digits[i]
        return int("".join(digits))
```

- Recommendation by sorting
```
class Solution:
    def largestInteger(self, num: int) -> int:
        l = [int(d) for d in list(str(num))]
        n = len(l)
        for i in range(n - 1):
            for j in range(i + 1, n):
                if (l[i] - l[j]) % 2 == 0 and l[i] < l[j]:
                    l[i], l[j] = l[j], l[i]
        return int("".join(str(d) for d in l))
        
```
