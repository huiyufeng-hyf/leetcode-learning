public class Solution {
    public string DigitSum(string s, int k) {
        while (s.Length > k)
        {
            var sb = new StringBuilder();
            for (var i = 0; i < s.Length; i += k)
            {
                var sum = 0;
                var end = Math.Min(i + k, s.Length);
                for (var j = i; j < end; j++)
                    sum += s[j] - '0';
                sb.Append(sum);
            }
            s = sb.ToString();
        }
        return s;
    }
}

**Python**
```
class Solution:
    def digitSum(self, s: str, k: int) -> str:
        while len(s) > k:
            temp = []
            for i in range(0, len(s), k):
                end = min(i + k, len(s))
                temp.append(str(sum(int(d) for d in s[i:end])))
            s = "".join(temp)
        return s
```
