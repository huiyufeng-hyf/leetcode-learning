**https://leetcode.cn/problems/find-the-k-beauty-of-a-number/description/**

- Solution by simulation
```
public class Solution {
    public int DivisorSubstrings(int num, int k) {
        var numStr = num.ToString();
        var n = numStr.Length;
        var ans = 0;
        for (var i = 0; i < n; i++)
        {
            if (i + k > n)
                break;
            var tmp = numStr.Substring(i, k);
            var d = Convert.ToInt32(tmp);
            if (d != 0 && num % d == 0)
                ans++;
        }
        return ans;
    }
}
```
**Python**
```
class Solution:
    def divisorSubstrings(self, num: int, k: int) -> int:
        num_str = str(num)
        n, ans = len(num_str), 0
        for i in range(n):
            if i + k > n:
                break
            tmp = num_str[i:i+k]
            d = int(tmp)
            if d != 0 and num % d == 0:
                ans += 1
        return ans
```

- Recommendation by enumeration
```
class Solution:
    def divisorSubstrings(self, num: int, k: int) -> int:
        s = str(num)
        n, res = len(s), 0
        for i in range(n - k + 1):
            tmp = int(s[i:i+k])
            if tmp != 0 and num % tmp == 0:
                res += 1
        return res
```
